Database Schemas Overview

All table schemas broken down by MVP phase.

MVP1 CORE

\n

system_info

CREATE TABLE system_info (\n    id INT PRIMARY KEY DEFAULT 1,\n    system_version VARCHAR(50) NOT NULL,\n    database_version VARCHAR(50) NOT NULL,\n    license_key VARCHAR(255),\n    installed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP\n);

company_settings

CREATE TABLE company_settings (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    setting_key VARCHAR(100) NOT NULL UNIQUE,\n    value TEXT NOT NULL,\n    type VARCHAR(20) NOT NULL DEFAULT 'string',\n    description TEXT,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP\n);

currencies

CREATE TABLE currencies (\n    code VARCHAR(10) PRIMARY KEY,\n    name VARCHAR(50) NOT NULL,\n    symbol VARCHAR(10) NOT NULL,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

exchange_rates

CREATE TABLE exchange_rates (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    base_currency VARCHAR(10) NOT NULL,\n    target_currency VARCHAR(10) NOT NULL,\n    rate DECIMAL(15,6) NOT NULL,\n    effective_date DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    created_by CHAR(36)\n);

branches

CREATE TABLE branches (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    name VARCHAR(150) NOT NULL,\n    code VARCHAR(20),\n    address TEXT,\n    city VARCHAR(100),\n    country VARCHAR(100) NOT NULL DEFAULT 'Cambodia',\n    phone VARCHAR(20),\n    email VARCHAR(150),\n    logo_url TEXT,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP,\n    deleted_by CHAR(36),\n    deleted_at DATETIME NULL\n);

branch_settings

CREATE TABLE branch_settings (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36) NOT NULL UNIQUE,\n    receipt_footer TEXT,\n    tax_rate DECIMAL(5,2),\n    default_language VARCHAR(10) NOT NULL DEFAULT 'en',\n    currency_code VARCHAR(10) NOT NULL DEFAULT 'USD',\n    timezone VARCHAR(50) NOT NULL DEFAULT 'Asia/Phnom_Penh',\n    business_hours JSON NOT NULL DEFAULT ('{}'),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP\n);

roles

CREATE TABLE roles (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    name VARCHAR(50) NOT NULL UNIQUE,\n    display_name VARCHAR(100) NOT NULL,\n    description TEXT,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP,\n    deleted_by CHAR(36),\n    deleted_at DATETIME NULL\n);

permissions

CREATE TABLE permissions (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    name VARCHAR(100) NOT NULL UNIQUE,\n    group_name VARCHAR(50),\n    description TEXT,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

role_permissions

CREATE TABLE role_permissions (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    role_id CHAR(36) NOT NULL,\n    permission_id CHAR(36) NOT NULL,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    UNIQUE(role_id, permission_id)\n);

users

CREATE TABLE users (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    role_id CHAR(36) NOT NULL,\n    username VARCHAR(50) NOT NULL,\n    first_name VARCHAR(75) NOT NULL,\n    last_name VARCHAR(75) NOT NULL,\n    email VARCHAR(150) NOT NULL,\n    phone VARCHAR(20),\n    password_hash VARCHAR(255) NOT NULL,\n    avatar_url TEXT,\n    gender ENUM('male', 'female', 'other'),\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP,\n    deleted_by CHAR(36),\n    deleted_at DATETIME NULL\n);

user_permissions

CREATE TABLE user_permissions (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    user_id CHAR(36) NOT NULL,\n    permission_id CHAR(36) NOT NULL,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    UNIQUE(user_id, permission_id)\n);

categories

CREATE TABLE categories (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    parent_id CHAR(36),\n    name VARCHAR(100) NOT NULL,\n    name_kh VARCHAR(100),\n    sort_order INT NOT NULL DEFAULT 0,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP,\n    deleted_by CHAR(36),\n    deleted_at DATETIME NULL\n);

products

CREATE TABLE products (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    category_id CHAR(36),\n    name VARCHAR(150) NOT NULL,\n    name_kh VARCHAR(150),\n    sku VARCHAR(100),\n    barcode VARCHAR(100),\n    selling_price DECIMAL(12,2) NOT NULL,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_by CHAR(36),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    updated_by CHAR(36),\n    updated_at DATETIME NULL ON UPDATE CURRENT_TIMESTAMP,\n    deleted_by CHAR(36),\n    deleted_at DATETIME NULL\n);

modifier_groups

CREATE TABLE modifier_groups (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    name VARCHAR(50) NOT NULL,\n    min_select INT NOT NULL DEFAULT 0,\n    max_select INT NOT NULL DEFAULT 1,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

modifier_options

CREATE TABLE modifier_options (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    modifier_group_id CHAR(36) NOT NULL,\n    value VARCHAR(100) NOT NULL,\n    price_delta DECIMAL(10,2) NOT NULL DEFAULT 0,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

product_modifier_mapping

CREATE TABLE product_modifier_mapping (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    product_id CHAR(36) NOT NULL,\n    modifier_group_id CHAR(36) NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    UNIQUE(product_id, modifier_group_id)\n);

tables

CREATE TABLE tables (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    number VARCHAR(20) NOT NULL,\n    capacity INT NOT NULL DEFAULT 2,\n    status ENUM('available', 'occupied', 'cleaning') NOT NULL DEFAULT 'available',\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

table_merges

CREATE TABLE table_merges (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    primary_table_id CHAR(36) NOT NULL,\n    merged_table_id CHAR(36) NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

customers

CREATE TABLE customers (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    first_name VARCHAR(75) NOT NULL,\n    last_name VARCHAR(75),\n    phone VARCHAR(20),\n    email VARCHAR(150),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

shifts

CREATE TABLE shifts (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    user_id CHAR(36) NOT NULL,\n    opening_cash DECIMAL(12,2) NOT NULL DEFAULT 0,\n    closing_cash DECIMAL(12,2),\n    status ENUM('open', 'closed') NOT NULL DEFAULT 'open',\n    opened_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,\n    closed_at DATETIME\n);

shift_cash_movements

CREATE TABLE shift_cash_movements (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    shift_id CHAR(36) NOT NULL,\n    type ENUM('in', 'out') NOT NULL,\n    amount DECIMAL(12,2) NOT NULL,\n    note TEXT,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

orders

CREATE TABLE orders (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    order_number VARCHAR(30) NOT NULL UNIQUE,\n    branch_id CHAR(36),\n    shift_id CHAR(36),\n    user_id CHAR(36) NOT NULL,\n    customer_id CHAR(36),\n    table_id CHAR(36),\n    parent_order_id CHAR(36),\n    split_type ENUM('none', 'even', 'item') NOT NULL DEFAULT 'none',\n    guest_count INT NOT NULL DEFAULT 1,\n    order_type ENUM('dine_in', 'takeaway', 'delivery', 'drive_thru') NOT NULL DEFAULT 'dine_in',\n    status ENUM('pending', 'confirmed', 'preparing', 'ready', 'completed', 'cancelled', 'voided') NOT NULL DEFAULT 'pending',\n    subtotal DECIMAL(12,2) NOT NULL DEFAULT 0,\n    tax_amount DECIMAL(12,2) NOT NULL DEFAULT 0,\n    total DECIMAL(12,2) NOT NULL DEFAULT 0,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

order_items

CREATE TABLE order_items (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    order_id CHAR(36) NOT NULL,\n    product_id CHAR(36),\n    product_name VARCHAR(150) NOT NULL,\n    quantity INT NOT NULL DEFAULT 1,\n    unit_price DECIMAL(10,2) NOT NULL,\n    line_total DECIMAL(12,2) NOT NULL DEFAULT 0,\n    selected_modifiers JSON NOT NULL DEFAULT ('[]'),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

taxes

CREATE TABLE taxes (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    name VARCHAR(50) NOT NULL,\n    rate DECIMAL(5,2) NOT NULL,\n    type ENUM('exclusive', 'inclusive') NOT NULL DEFAULT 'exclusive',\n    applies_to ENUM('all', 'dine_in', 'takeaway', 'delivery') NOT NULL DEFAULT 'all',\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

order_taxes

CREATE TABLE order_taxes (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    order_id CHAR(36) NOT NULL,\n    tax_name VARCHAR(50) NOT NULL,\n    rate DECIMAL(5,2) NOT NULL,\n    type VARCHAR(20) NOT NULL,\n    amount DECIMAL(12,2) NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

sales

CREATE TABLE sales (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    sale_number VARCHAR(30) NOT NULL UNIQUE,\n    order_id CHAR(36) NOT NULL,\n    branch_id CHAR(36),\n    shift_id CHAR(36),\n    user_id CHAR(36) NOT NULL,\n    customer_id CHAR(36),\n    subtotal DECIMAL(12,2) NOT NULL,\n    tax_amount DECIMAL(12,2) NOT NULL DEFAULT 0,\n    total DECIMAL(12,2) NOT NULL,\n    sale_date DATE NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

sale_items

CREATE TABLE sale_items (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    sale_id CHAR(36) NOT NULL,\n    product_id CHAR(36),\n    product_name VARCHAR(150) NOT NULL,\n    quantity INT NOT NULL,\n    unit_price DECIMAL(10,2) NOT NULL,\n    line_total DECIMAL(12,2) NOT NULL,\n    selected_modifiers JSON NOT NULL DEFAULT ('[]'),\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

sale_taxes

CREATE TABLE sale_taxes (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    sale_id CHAR(36) NOT NULL,\n    tax_name VARCHAR(50) NOT NULL,\n    rate DECIMAL(5,2) NOT NULL,\n    type VARCHAR(20) NOT NULL,\n    amount DECIMAL(12,2) NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

payment_methods

CREATE TABLE payment_methods (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    name VARCHAR(100) NOT NULL,\n    type ENUM('cash', 'card', 'qr', 'ewallet', 'credit', 'gift_card', 'bank_transfer') NOT NULL DEFAULT 'cash',\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

sale_payments

CREATE TABLE sale_payments (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    sale_id CHAR(36) NOT NULL,\n    payment_method_id CHAR(36) NOT NULL,\n    amount DECIMAL(12,2) NOT NULL,\n    amount_tendered DECIMAL(12,2),\n    change_amount DECIMAL(12,2) NOT NULL DEFAULT 0,\n    currency_code VARCHAR(10) NOT NULL DEFAULT 'USD',\n    exchange_rate DECIMAL(10,4) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

coupons

CREATE TABLE coupons (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    code VARCHAR(50) NOT NULL,\n    type ENUM('percent', 'fixed') NOT NULL,\n    value DECIMAL(10,2) NOT NULL,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

coupon_usages

CREATE TABLE coupon_usages (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    coupon_id CHAR(36) NOT NULL,\n    order_id CHAR(36),\n    customer_id CHAR(36),\n    discount_amount DECIMAL(10,2) NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

delivery_platforms

CREATE TABLE delivery_platforms (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    name VARCHAR(100) NOT NULL,\n    commission_pct DECIMAL(5,2) NOT NULL DEFAULT 0,\n    is_active TINYINT(1) NOT NULL DEFAULT 1,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

delivery_orders

CREATE TABLE delivery_orders (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    order_id CHAR(36),\n    platform_id CHAR(36),\n    platform_order_id VARCHAR(100),\n    driver_name VARCHAR(100),\n    driver_phone VARCHAR(20),\n    delivery_fee DECIMAL(10,2) NOT NULL DEFAULT 0,\n    status ENUM('pending', 'confirmed', 'preparing', 'ready', 'picked_up', 'delivered', 'cancelled') NOT NULL DEFAULT 'pending',\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

expense_categories

CREATE TABLE expense_categories (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    name VARCHAR(100) NOT NULL UNIQUE,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

expenses

CREATE TABLE expenses (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36),\n    category_id CHAR(36) NOT NULL,\n    shift_id CHAR(36),\n    amount DECIMAL(12,2) NOT NULL,\n    expense_date DATE NOT NULL,\n    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP\n);

daily_summary

CREATE TABLE daily_summary (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36) NOT NULL,\n    summary_date DATE NOT NULL,\n    total_orders INT NOT NULL DEFAULT 0,\n    total_sales DECIMAL(14,2) NOT NULL DEFAULT 0,\n    UNIQUE(branch_id, summary_date)\n);

daily_product_summary

CREATE TABLE daily_product_summary (\n    id CHAR(36) PRIMARY KEY DEFAULT (UUID()),\n    branch_id CHAR(36) NOT NULL,\n    summary_date DATE NOT NULL,\n    product_id CHAR(36),\n    qty_sold INT NOT NULL DEFAULT 0,\n    total_revenue DECIMAL(14,2) NOT NULL DEFAULT 0,\n    UNIQUE(branch_id, summary_date, product_id)\n);
